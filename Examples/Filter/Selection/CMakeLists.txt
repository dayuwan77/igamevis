foreach(name IN ITEMS TestExtractSelection ExtractionExamples)
    if(NOT TARGET ${name})
        add_executable(${name} ${name}.cpp)
        target_link_libraries(${name} PRIVATE iGameCore)
        if(MSVC)
            target_compile_options(${name} PRIVATE /EHsc)
        endif()
        add_test(NAME ${name} COMMAND ${name})
        set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${IGAMEVIS_ROOT_DIR}")
    endif()
endforeach()
